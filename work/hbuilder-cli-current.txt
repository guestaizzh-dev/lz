
--------------- HBuilderX(v5.26.2026091802) cli (v1.0.0.0)--------------
cloud functions	

  云开发

  参数:

	--prj   	项目名称或项目在列表中的编号

	--provider	云服务提供商代号, tcb|aliyun|alipay 取值有[tcb|aliyun|alipay]

	--name  	资源名称

	--unimod	资源所属uni_module模块名称

	--cloud 	获取云端资源，无值

	--list  	获取资源列表
	可为 cloudfunction、common、db、vf、action 或 space
	此时如果填入参数unimod，则操作uni_module模块下资源

	获取云函数列表
	本地:cli cloud functions --list cloudfunction --prj projectname --provider aliyun [--unimod modname]
	云端:cli cloud functions --list cloudfunction --prj projectname --provider aliyun --cloud [--unimod modname]

	获取云函数公共模块列表
	本地:cli cloud functions --list common --prj projectname --provider aliyun [--unimod modname]
	云端:cli cloud functions --list common --prj projectname --provider aliyun --cloud [--unimod modname]

	获取数据库集合Schema列表
	本地:cli cloud functions --list db --prj projectname --provider aliyun [--unimod modname]
	云端:cli cloud functions --list db --prj projectname --provider aliyun --cloud [--unimod modname]

	获取数据库校验函数列表
	本地:cli cloud functions --list vf --prj projectname --provider aliyun [--unimod modname]
	云端:cli cloud functions --list vf --prj projectname --provider aliyun --cloud [--unimod modname]

	获取action列表
	本地:cli cloud functions --list action --prj projectname --provider aliyun [--unimod modname]
	云端:cli cloud functions --list action --prj projectname --provider aliyun --cloud [--unimod modname]

	获取云空间列表
	cli cloud functions --list space --prj projectname --provider aliyun


	--download	下载项目资源
	可为 cloudfunction、common、db、vf、action 或 all
	除aallcloudfunctions外此时如果填入参数unimod，则操作uni_module模块下资源
	action不接收资源名称的参数

	下载某个云函数
	cli cloud functions --download cloudfunction --prj projectname --provider aliyun --name cloudfunctionname [--unimod modname]

	下载某个云函数公共模块
	cli cloud functions --download common --prj projectname --provider aliyun --name commonname [--unimod modname]

	下载某个数据库集合Schema
	cli cloud functions --download db --prj projectname --provider aliyun --name databasename [--unimod modname]

	下载某个数据库校验函数
	cli cloud functions --download vf --prj projectname --provider aliyun --name validateFunctionname [--unimod modname]

	下载整个client db action
	cli cloud functions --download action --prj projectname --provider aliyun [--unimod modname]

	下载所有云函数、云函数公共模块及actions
	cli cloud functions --download all --prj projectname --provider aliyun [--unimod modname]


	--upload	上传项目资源
	可为 cloudfunction、common、db、vf、action 或 all
	除all外此时如果填入参数unimod，则操作uni_module模块下资源

	上传某个云函数
	cli cloud functions --upload cloudfunction --prj projectname --provider aliyun --name cloudfunctionname [--unimod modname]

	上传某个云函数公共模块
	cli cloud functions --upload common --prj projectname --provider aliyun --name commonname [--unimod modname]

	上传某个数据库集合Schema
	cli cloud functions --upload db --prj projectname --provider aliyun --name databasename [--unimod modname]

	上传某个数据库校验函数
	cli cloud functions --upload vf --prj projectname --provider aliyun --name validateFunctionname [--unimod modname]

	上传某个action
	cli cloud functions --upload action --prj projectname --provider aliyun --name cdbname [--unimod modname]

	上传所有云函数、云函数公共模块及actions
	cli cloud functions --upload allcloudfunctions --prj projectname --provider aliyun [--unimod modname]



	--initdatabase	初始化数据库数

	--assignspace	指定云空间

	--run   	运行云函数 可为 cloudfunction、cloudobject 
	运行项目中支付宝云服务空间云函数
	cli cloud functions --run cloudfunction --prj projectname --provider alipay --name cloudfunctionname

	运行项目中支付宝云服务空间云对象方法
	cli cloud functions --run cloudobject --prj projectname --provider alipay --name cloudobjectname 


	--create	创建云开发环境；可为 env、cloudfunction、cloudoobject、commonmodule
	创建支付宝云服务空间环境
	cli cloud functions --create env --prj projectname --provider alipay
	创建云函数
	cli cloud functions --create cloudfunction --prj projectname --provider alipay --name cfname [--unimod modname]
	创建云对象
	cli cloud functions --create cloudobject --prj projectname --provider alipay --name coname [--unimod modname]
	创建云对象
	cli cloud functions --create commonmodule --prj projectname --provider alipay --name commonname [--unimod modname]


	--info  	获取项目云开发环境信息


compile app-android	

  通过 CLI 编译 Android 平台 UTS 单文件或单 uni_modules 模块

  用法：./cli compile app-android --help
        ./cli compile app-android --project demo --file src/uni_modules/demo/utssdk/app-android/index.uts
        ./cli compile app-android --project demo --uni_module aa-bb
        说明：--file 与 --uni_module 不能同时指定；--uni_module 可传绝对路径、相对路径，也可只传模块名（如 aa-bb，会自动拼接为项目源码根目录下的 uni_modules/aa-bb）

  参数:

	--help  	查看 UTS 编译 CLI 命令帮助

	--project	HBuilderX 中已导入项目的名称或绝对路径

	--file  	要编译的 UTS 文件路径，支持绝对路径或相对项目目录的路径；与 --uni_module 互斥

	--uni_module	要编译的 uni_modules 模块目录，支持绝对路径、相对项目目录路径或模块名简写；与 --file 互斥

launch app-android	

  通过 CLI 运行到 Android App。文档：https://hx.dcloud.net.cn/cli/launch-app?id=launch-app-android

  用法：./cli launch app-android --help

  参数:

	--help  	查看运行 CLI 命令帮助

	--project	HBuilderX 中已导入项目的名称或绝对路径

	--ui    	是否切换为界面运行模式；默认使用纯 CLI 运行，传入 --ui 后通过界面方式启动 取值有[true|false] 默认值为:false

	--deviceId	指定要使用的设备序列号；不指定时默认使用第一个设备，可通过 devices list 获取设备列表

	--playground	基座类型；custom 时可通过 customType 选择本地基座或已安装基座 取值有[standard|custom] 默认值为:standard

	--customType	自定义基座类型；local 为本地基座，device 为设备上已安装的基座（需 playground=custom） 取值有[local|device] 默认值为:local

	--packageName	设备上已安装基座的 Android 包名（customType=device 时必填）

	--localRepoPath	已安装基座关联的 Android 原生工程路径（customType=device 时可选）

	--native-log	是否显示原生日志 取值有[true|false] 默认值为:false

	--cleanCache	是否清理构建缓存（仅 uni-app x 项目生效） 取值有[true|false] 默认值为:false

	--vaporRenderTarget	视图层编译目标，可选 bytecode 或 nativecode（仅 uni-app x 蒸汽模式生效） 取值有[bytecode|nativecode] 默认值为:bytecode

	--compile	是否以编译模式运行，仅编译代码 取值有[true|false] 默认值为:false

	--continue-on-error	编译出错后是否继续运行 取值有[true|false] 默认值为:false

	--runtime-log-format	运行时日志输出格式；formatted 为现有格式化日志，raw 为设备原始结构化日志 取值有[formatted|raw] 默认值为:formatted

	--pagePath	启动页面路径

	--pageQuery	启动页面查询参数，多个参数使用 & 连接
